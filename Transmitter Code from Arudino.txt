"use client";

import React, { useState } from "react";

// Must match the UUIDs in your ArduinoBLE sketch
const SERVICE_UUID = "19b10000-e8f2-537e-4f6c-d104768a1214";
const CHAR_UUID    = "19b10001-e8f2-537e-4f6c-d104768a1214";

export default function TableBluetoothConnect({ tableId = "table-1" }: { tableId?: string }) {
  const [isConnected, setIsConnected] = useState(false);
  const [isOccupied, setIsOccupied]   = useState<boolean | null>(null);
  const [deviceName, setDeviceName]   = useState<string>("");
  const [error, setError]             = useState<string>("");

  // Optional: Sync the live Bluetooth state straight to your Tiger Data / Next.js backend
  const syncWithBackend = async (occupied: boolean) => {
    try {
      await fetch("/api/readings", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          table_id: tableId,
          occupied: occupied,
          timestamp: new Date().toISOString(),
        }),
      });
    } catch (err) {
      console.error("Backend sync error:", err);
    }
  };

  const connectToTable = async () => {
    setError("");
    try {
      if (!("bluetooth" in navigator)) {
        throw new Error("Web Bluetooth is not supported in this browser. Use Chrome on Android/Desktop or Bluefy on iOS.");
      }

      // 1. Scan for the Arduino UNO R4 WiFi BLE device
      const device = await (navigator as any).bluetooth.requestDevice({
        filters: [{ namePrefix: "GhostSeat" }],
        optionalServices: [SERVICE_UUID],
      });

      setDeviceName(device.name || "GhostSeat-Table1");

      device.addEventListener("gattserverdisconnected", () => {
        setIsConnected(false);
        setIsOccupied(null);
      });

      // 2. Connect to GATT Server & Service
      const server = await device.gatt.connect();
      const service = await server.getPrimaryService(SERVICE_UUID);
      const characteristic = await service.getCharacteristic(CHAR_UUID);

      // 3. Read the initial IR sensor state (1 = Occupied/Green, 0 = Open/Blue)
      const initialValue = await characteristic.readValue();
      const initialOccupied = initialValue.getUint8(0) === 1;
      setIsOccupied(initialOccupied);
      setIsConnected(true);
      syncWithBackend(initialOccupied);

      // 4. Subscribe to real-time live notifications whenever the IR sensor changes!
      await characteristic.startNotifications();
      characteristic.addEventListener("characteristicvaluechanged", (event: any) => {
        const value = event.target.value.getUint8(0);
        const occupiedState = value === 1;
        setIsOccupied(occupiedState);
        syncWithBackend(occupiedState);
      });

    } catch (err: any) {
      setError(err.message || "Failed to connect to table via Bluetooth.");
    }
  };

  return (
    <div className="p-6 max-w-md mx-auto rounded-2xl border border-slate-800 bg-slate-900 text-white shadow-xl">
      <div className="flex items-center justify-between mb-4">
        <h2 className="text-xl font-bold">Table Bluetooth Link</h2>
        <span className={`px-3 py-1 rounded-full text-xs font-semibold ${
          isConnected ? "bg-emerald-500/20 text-emerald-400" : "bg-slate-700 text-slate-400"
        }`}>
          {isConnected ? `Connected: ${deviceName}` : "Disconnected"}
        </span>
      </div>

      {!isConnected ? (
        <button
          onClick={connectToTable}
          className="w-full py-3 px-4 bg-indigo-600 hover:bg-indigo-500 active:bg-indigo-700 rounded-xl font-semibold transition"
        >
          Pair with Table Sensor (BLE)
        </button>
      ) : (
        <div className="p-4 rounded-xl bg-slate-800/80 border border-slate-700 text-center">
          <p className="text-sm text-slate-400 mb-1">Live IR Sensor Status</p>
          <div className="flex items-center justify-center gap-3 mt-2">
            <span className={`h-4 w-4 rounded-full ${
              isOccupied ? "bg-green-500 shadow-[0_0_12px_#22c55e]" : "bg-blue-500 shadow-[0_0_12px_#3b82f6]"
            }`} />
            <span className="text-2xl font-bold">
              {isOccupied ? "OCCUPIED (True)" : "OPEN (False)"}
            </span>
          </div>
          <p className="text-xs text-slate-400 mt-2">
            {isOccupied 
              ? "Green LED ON — Student detected in range!" 
              : "Blue LED ON — Seat is available."}
          </p>
        </div>
      )}

      {error && (
        <p className="mt-3 text-xs text-red-400 bg-red-950/40 p-2 rounded-lg border border-red-800/50">
          {error}
        </p>
      )}
    </div>
  );
}
